# Build only. Loading/installing the candidate requires maintainer approval.
cmake_minimum_required(VERSION 3.25)
project(joetoo_managed_close VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core DBus Widgets)
include(KDEInstallDirs6)
option(JOETOO_KWIN_X11 "Build for the separate kwin-x11 compositor" OFF)
if(JOETOO_KWIN_X11)
    find_package(KWinX11 6.7.6 EXACT REQUIRED)
    set(kwin_target KWinX11::kwin)
    set(effect_namespace kwin-x11)
    set(compositor_binary kwin_x11)
else()
    find_package(KWin 6.7.6 EXACT REQUIRED)
    set(kwin_target KWin::kwin)
    set(effect_namespace kwin)
    set(compositor_binary kwin_wayland)
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(XCB REQUIRED IMPORTED_TARGET xcb)
add_executable(joetoo-kwin-close-x11 x11.cpp)
target_link_libraries(joetoo-kwin-close-x11 PRIVATE PkgConfig::XCB)
add_library(joetoo_managed_close MODULE adapter.cpp)
set_target_properties(joetoo_managed_close PROPERTIES PREFIX "")
target_link_libraries(joetoo_managed_close PRIVATE ${kwin_target} Qt6::DBus PkgConfig::XCB)
target_compile_definitions(joetoo_managed_close PRIVATE
    JOETOO_COMPOSITOR_BINARY="${compositor_binary}"
    JOETOO_CLOSE_HELPER="${KDE_INSTALL_FULL_LIBEXECDIR}/joetoo-kwin-close-x11")
install(TARGETS joetoo_managed_close DESTINATION ${KDE_INSTALL_PLUGINDIR}/${effect_namespace}/effects/plugins)
install(TARGETS joetoo-kwin-close-x11 DESTINATION ${KDE_INSTALL_LIBEXECDIR})
